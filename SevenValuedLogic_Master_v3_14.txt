\section{Перекрёстный вывод базиса $\{3, 4\}$ и семизначной алгебры}
% ==============================================================================

Пусть $S_n$ — симметрическая группа с равномерной инвариантной мерой Хаара $P(\sigma) = 1/n!$. Числа встреч $D_{n,j} = \binom{n}{j}\subfact{(n-j)}$ подсчитывают перестановки ровно с $j$ неподвижными точками; субфакториал $\subfact{n} = D_{n,0}$ задаёт мощность полных деранжментов $M_0(n)$ (динамический ноль), а $\Delta_n = n! - \subfact{n}$ — число инвариантно-несущих конфигураций $M_1(n)$, удовлетворяющее точному рекуррентному соотношению $\Delta_{n+1} = (n+1)\Delta_n + (-1)^n$.

\begin{definition}[Конструктивные координатные биекции для $n = 3$ и $n = 4$]
Поскольку в $S_3$ оба полных деранжмента $M_0(3) = \{(123), (132)\}$ являются чётными перестановками и лежат в $A_3$, поле $\mathbb{F}_2$ вводится конструктивно через фиксацию ориентации $3$-цикла: $\phi_3((123)) = 0$, $\phi_3((132)) = 1$. Аналогично, поскольку множество $M_1(4)$ мощности $\Delta_4 = 15$ не образует подгруппы в $S_4$ ($16 \nmid 24$), для линейного кодирования задаётся явная биекция $\iota: M_1(4) \sqcup \{e\} \to \mathbb{F}_2^4$, переводящая тождественную перестановку $e$ в нулевой вектор $0000$, а $15$ элементов $M_1(4)$ — в $15$ ненулевых векторов $\mathbb{F}_2^4 \setminus \{0\} \cong \mathrm{PG}(3,2)$.
\end{definition}

\begin{theorem}[Тройная фиксация узла $n = 3$ и базиса $\{3, 4\}$]
Узел $n = 3$ и переход $3 \to 4$ выделяются на множестве натуральных чисел пересечением трёх независимых утверждений:
\begin{enumerate}
    \item \textbf{Единственность простого субфакториала и поля регистрации:} Из рекуррентного тождества $\subfact{n} = (n-1)\bigl(\subfact{(n-1)} + \subfact{(n-2)}\bigr)$ следует, что $\subfact{3} = 2$ — единственное простое число среди всех субфакториалов ($\subfact{1} = 0$, $\subfact{2} = 1$, а при $n \ge 4$ оба сомножителя строго больше $1$). Линейное синдромное кодирование требует конечного поля $\mathbb{F}_{\subfact{n}}$: при $n = 3$ возникает $\mathbb{F}_2$; при $n = 4$ — $\mathbb{F}_9$ ($\subfact{4} = 9 = 3^2$); при $n = 5$ величина $\subfact{5} = 44 = 4 \cdot 11$ не является степенью простого числа, поэтому поля $\mathbb{F}_{44}$ не существует.
    \item \textbf{Диофантово насыщение притока конфигураций:} При переходе $n \to n+1$ число нетривиальных направленных корреляционных каналов в $(n+1)$-элементном базисе равно $(n+1)^2 - (n+1) - 1 = n(n+1) - 1$. Диофантово уравнение $\Delta_{n+1} - \Delta_n = n(n+1) - 1$ на области $n \ge 3$ имеет единственный натуральный корень $n = 3$, где $\Delta_4 - \Delta_3 = 15 - 4 = 11 = 3 \cdot 4 - 1$ (при $n = 4$ имеем $76 - 15 = 61 \neq 19$, при $n \ge 5$ выполняется $\Delta_{n+1} - \Delta_n \ge 5(n+3) - 1 > n(n+1) - 1$).
    \item \textbf{Совершенство кода Хэмминга $[7,4,3]_2$:} Защита $k_{\mathrm{info}} = 4$ информационных бит ($\Delta_4 = 15$ ненулевых состояний и $1$ нулевое состояние $0000$) от одиночных ошибок над $\mathbb{F}_2$ по границе Хэмминга $2^r \ge 4 + r + 1$ фиксирует минимальную ширину синдрома $r = 3$ ($2^3 = 8 = 4 + 3 + 1$) и длину блока $n_{\mathrm{code}} = 4 + 3 = 7$.
\end{enumerate}
\end{theorem}

\begin{definition}[Семизначная квазигруппа Штейнера и октонионы]
Пространство $3$-битных синдромов расщепляется как $\mathbb{F}_2^3 = \{000\} \sqcup \mathrm{PG}(2,2)$ ($8 = 1 + 7$). На семи ненулевых синдромах операция инцидентности $e_i \star e_j = e_k$ (третья точка прямой через $e_i, e_j$ при $i \neq j$ и $e_i \star e_i = e_i$, алгебраически $a \star b = a \oplus b$ при $a \neq b$) образует замкнутую коммутативную идемпотентную квазигруппу Штейнера с внешним аннулятором $000$. Фиксация циклической ориентации на $7$ прямых плоскости Фано задаёт структурные константы $c_{ijk} \in \{-1, 0, +1\}$ единственной $8$-мерной нормированной алгебры с делением Гурвица — алгебры октонионов $\mathbb{O} = \mathbb{R}e_0 \oplus \mathrm{Im}(\mathbb{O})$.
\end{definition}

\begin{theorem}[Несуществование стационара и стрела времени]
При $n = 3$ и ширине синдрома $r = 3$ однотактный приток конфигураций ($\delta\Delta_3 = 11$) строго превышает ёмкость ненулевых синдромов ($2^3 - 1 = 7 < 11$). Однотактное стационарное вложение невозможно, что вынуждает двухтактный обмен в $7$-кубитном гильбертовом пространстве $\mathcal{H}_{\mathrm{tot}} = (\mathbb{C}^2)^{\otimes 7}$ ($\dim = 128$, разбитом на секторы $\Pi_3$ ранга $8$, $\Pi_4$ ранга $15$ и $\Pi_\perp$ ранга $105$):
\begin{enumerate}
    \item \textbf{Прямой канал ($S_3 \to S_4$):} Каноническая цикловая вставка $\sigma = (i\,4) \circ \pi^{(4)}$ ($i \in \{1..4\}$, $\pi \in M_1(3)$) порождает $4 \times 4 = 16$ конфигураций. Ровно $3$ из них теряют неподвижную точку — это три полных паросочетания четвёрки $(12)(34), (13)(24), (14)(23)$, давая $W_{\mathrm{fwd}}(2) = 13/16$.
    \item \textbf{Обратный канал ($S_4 \to S_3$):} Цикловое сжатие элемента $4$ из $\Delta_4 = 15$ конфигураций $S_4$ сохраняет неподвижную точку в $13$ случаях, а $2$ чистых $3$-цикла $(123)(4)$ и $(132)(4)$ сбрасываются в $M_0(3)$ ($\subfact{3} = 2$), давая $W_{\mathrm{bwd}}(2) = 13/15$.
    \item \textbf{Нарушение детального баланса:} Коэффициент асимметрии равен $\chi(2) = W_{\mathrm{bwd}}(2)/W_{\mathrm{fwd}}(2) = 16/15 > 1$, квант дефекта $\delta_{\mathrm{asym}}(2) = 13/240$, а стационарное распределение цепи равно $\pi = (16/31, 15/31)$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Масштабирование $\mathrm{PG}(2,q)$, селекция $q = 2$ и положительность $T^2(q)$]
При масштабировании на конечное поле $\mathbb{F}_q$ ($q = p^m \ge 2$) фактор-пространство $(\mathbb{F}_q^3 \setminus \{0\})/\mathbb{F}_q^*$ изоморфно $\mathrm{PG}(2,q)$ с $N(q) = q^2 + q + 1$ точками и тактовым периодом $T_0(q) = 2(N(q) - 1) = 2q(q + 1)$ ($T_0(2) = 12 = |A_4|$). Погружение $\mathrm{PG}(2, q+1)$ в аффинную плоскость $\mathrm{AG}(2, q+2)$ и $\mathrm{AG}(2, q+2) \setminus \{0\}$ даёт:
\[
W_{\mathrm{fwd}}(q) = \frac{q^2 + 3q + 3}{(q + 2)^2}, \qquad W_{\mathrm{bwd}}(q) = \frac{q^2 + 3q + 3}{(q + 1)(q + 3)}, \qquad \chi(q) = 1 + \frac{1}{(q + 1)(q + 3)} > 1.
\]
\begin{enumerate}
    \item \textbf{Диофантова селекция $q \in \{2, 4\}$:} При калибровке удельного дефекта $\tilde{\delta}_{\mathrm{asym}}(q) = [(q+2)^2((q+2)^2 - 1)]^{-1}$, порога эквипартиции $C_{\mathrm{reset}}(q) = 1/(2q)$ и примитивного квадратичного базиса Фаульхабера $B(q) = 6\sum_{k=1}^q k^2 = q(q+1)(2q+1)$, условие целочисленности нормированного числа тактов $\tau_{\mathrm{life}}^{(2)}(q) = [C_{\mathrm{reset}}(q)/\tilde{\delta}_{\mathrm{asym}}(q)] \cdot T_0(q) / B(q) \in \mathbb{N}$ на всём бесконечном луче $q \ge 2$ сводится к делимости $q \mid 12$ и $(2q + 1) \mid 45$, что имеет ровно два решения: $q \in \{2, 4\}$. Простой корень $q = 2$ фиксирует $\tau_{\mathrm{life}}^{(2)}(2) = 720/30 = 24$ и полный вихрь Фано $\tau_{\mathrm{life}}^{(7)}(2) = 7 \times 24 = 168 = |\mathrm{PSL}(2,7)|$.
    \item \textbf{Отражательная положительность Остервальдера — Шрадера:} Одношаговая матрица перехода $T(q)$ самосопряжена в $L^2(\pi)$ в силу условия детального баланса $\pi_0 W_{\mathrm{fwd}}(q) = \pi_1 W_{\mathrm{bwd}}(q)$, но имеет отрицательное собственное значение $\lambda_2(q) = 1 - W_{\mathrm{fwd}}(q) - W_{\mathrm{bwd}}(q) < 0$ ($\lambda_2(2) = -163/240$). Её двухшаговая композиция $T^2(q)$ строго положительно определена в $L^2(\pi)$ со спектром $\{1, \lambda_2(q)^2\} \subset (0, \infty)$. Это гарантирует существование самосопряжённого гамильтониана $H_{\mathrm{eff}}(q) = -(2\Delta\tau_{\mathrm{metric}}(q))^{-1}\ln T^2(q)$ и корректное аналитическое продолжение Вика ($\tau \to it$).
\end{enumerate}
\end{theorem}

На торсоре $H_{12} \cong \mathbb{C}[A_4]$ однопараметрическая фазовая когерентность оператора сдвига $T_{\mathrm{cycle}}$ выделяет $\mathrm{U}(1)_Y$.
