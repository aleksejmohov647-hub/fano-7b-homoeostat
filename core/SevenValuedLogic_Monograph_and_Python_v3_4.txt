def iaaft_surrogate(x, max_iter=200, rng=None):
    """Generates one IAAFT surrogate preserving exact spectrum and amplitude histogram."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(x)
    sorted_x = np.sort(x)
    amp_spectrum = np.abs(np.fft.rfft(x))
    surr = rng.permutation(x)
    for _ in range(max_iter):
        phases = np.angle(np.fft.rfft(surr))
        s_time = np.fft.irfft(amp_spectrum * np.exp(1j * phases), n=n)
        ranks = np.argsort(np.argsort(s_time))
        new_surr = sorted_x[ranks]
        if np.array_equal(new_surr, surr):
            break
        surr = new_surr
    return surr

    N = len(x)
    if N < 5:
        raise ValueError("Time series must contain at least 5 observations.")

    p3_list = [ordinal_pattern_code(x[t:t+3]) for t in range(N - 2)]
    p4_list = [ordinal_pattern_code(x[t:t+4]) for t in range(N - 3)]

    s3_is_m0 = [is_derangement(p) for p in p3_list]
    s4_is_m0 = [is_derangement(p) for p in p4_list]

    p_m0_3 = np.mean(s3_is_m0)
    p_m0_4 = np.mean(s4_is_m0)
    pi_emp = np.array([0.5 * p_m0_3, 0.5 * (1.0 - p_m0_3),
                       0.5 * p_m0_4, 0.5 * (1.0 - p_m0_4)])
    pi_dual = np.array([1.0/6.0, 1.0/3.0, 3.0/16.0, 5.0/16.0])

    mask = pi_emp > 0.0
    d_kl = float(np.sum(pi_emp[mask] * np.log(pi_emp[mask] / pi_dual[mask])))

    fwd_00 = fwd_0_tot = fwd_11 = fwd_1_tot = 0
    bwd_00 = bwd_0_tot = bwd_11 = bwd_1_tot = 0

    for t in range(N - 3):
        m0_3_curr = s3_is_m0[t]
        m0_4_curr = s4_is_m0[t]
        m0_3_bwd = (is_derangement(contract3_perm(p4_list[t]))
                    if mode == "contract3" else s3_is_m0[t + 1])

        if m0_3_curr:
            fwd_0_tot += 1
            if m0_4_curr:
                fwd_00 += 1
        else:
            fwd_1_tot += 1
            if not m0_4_curr:
                fwd_11 += 1

        if m0_4_curr:
            bwd_0_tot += 1
            if m0_3_bwd:
                bwd_00 += 1
        else:
            bwd_1_tot += 1
            if not m0_3_bwd:
                bwd_11 += 1

    w_fwd_0 = fwd_00 / fwd_0_tot if fwd_0_tot > 0 else np.nan
    w_bwd_0 = bwd_00 / bwd_0_tot if bwd_0_tot > 0 else np.nan
    w_fwd_1 = fwd_11 / fwd_1_tot if fwd_1_tot > 0 else np.nan
    w_bwd_1 = bwd_11 / bwd_1_tot if bwd_1_tot > 0 else np.nan

    if np.isnan(w_fwd_0) or np.isnan(w_bwd_0) or w_bwd_0 == 0.0:
        chi_0 = np.inf if (not np.isnan(w_fwd_0) and w_fwd_0 > 0.0) else np.nan
    else:
        chi_0 = w_fwd_0 / w_bwd_0

    if np.isnan(w_fwd_1) or np.isnan(w_bwd_1) or w_fwd_1 == 0.0:
        chi_1 = np.inf if (not np.isnan(w_bwd_1) and w_bwd_1 > 0.0) else np.nan
    else:
        chi_1 = w_bwd_1 / w_fwd_1

    wf0_safe = 0.0 if np.isnan(w_fwd_0) else w_fwd_0
    wb0_safe = 0.0 if np.isnan(w_bwd_0) else w_bwd_0

    # Subtract exact geometric offset (-1/24) in Protocol A so equilibrium J_net = 0
    eq_flux_offset = (-1.0 / 24.0) if mode == "contract3" else 0.0
    j_net = abs((pi_emp[0] * wf0_safe - pi_emp[2] * wb0_safe) - eq_flux_offset)
    v7_state = quantize_7vl_state(wf0_safe, wb0_safe, chi_0, chi_1, mode=mode)

    return {
        "pi_emp": tuple(np.round(pi_emp, 4)),
        "chi_0": round(float(chi_0), 3) if np.isfinite(chi_0) else float(chi_0),
        "chi_1": round(float(chi_1), 3) if np.isfinite(chi_1) else float(chi_1),
        "D_KL": round(d_kl, 6),
        "J_net": round(float(j_net), 6),
        "7VL_state": v7_state,
        "7VL_label": V7_LABELS[v7_state]
    }

def evaluate_with_surrogates(x, mode="window", n_surrogates=200, seed=42, z_threshold=3.0):
    """Evaluates empirical series against IAAFT linear-spectrum null ensemble with Z-score verdict."""
    rng = np.random.default_rng(seed)
    obs = compute_7vl_metrics(x, mode=mode)
    surr_dkl, surr_jnet = [], []
    for _ in range(n_surrogates):
        s = iaaft_surrogate(x, rng=rng)
        m = compute_7vl_metrics(s, mode=mode)
        surr_dkl.append(m["D_KL"])
        surr_jnet.append(m["J_net"])
    std_dkl = np.std(surr_dkl)
    std_jnet = np.std(surr_jnet)
    z_kl = float((obs["D_KL"] - np.mean(surr_dkl)) / std_dkl) if std_dkl > 1e-12 else np.nan
    z_asym = float((obs["J_net"] - np.mean(surr_jnet)) / std_jnet) if std_jnet > 1e-12 else np.nan
    obs["Z_KL"] = round(z_kl, 2) if np.isfinite(z_kl) else np.nan
    obs["Z_asym"] = round(z_asym, 2) if np.isfinite(z_asym) else np.nan
    obs["nonlinear_irreversible"] = bool(
        np.isfinite(z_kl) and np.isfinite(z_asym) and (z_kl > z_threshold) and (z_asym > z_threshold)
    )
    return obs
