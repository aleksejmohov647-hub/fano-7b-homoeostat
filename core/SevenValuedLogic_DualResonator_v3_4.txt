/-- Type-safe cyclic contraction projecting Perm4 strictly onto Perm3 (deletes 3 from cycle). -/
def contract3 (m : Perm4) : Perm3 :=
  let step (x : Fin 4) : Fin 3 :=
    let y := m.eval x
    let z := if y == ⟨3, by decide⟩ then m.eval ⟨3, by decide⟩ else y
    ⟨z.val % 3, by omega⟩
  { f0 := step ⟨0, by decide⟩
    f1 := step ⟨1, by decide⟩
    f2 := step ⟨2, by decide⟩ }

/-- Ordinal window extension channels (Bandt-Pompe right-insertion at index 3):
    Inserting x_3 of rank r in {0,1,2,3} shifts elements >= r by +1. -/
def bpInsert (p : Perm3) (r : Fin 4) : Perm4 :=
  let lift (v : Fin 3) : Fin 4 :=
    if v.val < r.val then ⟨v.val, by omega⟩ else ⟨v.val + 1, by omega⟩
  { f0 := lift p.f0
    f1 := lift p.f1
    f2 := lift p.f2
    f3 := r }

/-- Ordinal window truncation (Bandt-Pompe left-drop of x_0):
    Dropping x_0 shifts ranks > f0 down by -1. -/
def bpDropLeft (m : Perm4) : Perm3 :=
  let drop (v : Fin 4) : Fin 3 :=
    if v.val < m.f0.val then ⟨v.val, by omega⟩ else ⟨(v.val - 1) % 3, by omega⟩
  { f0 := drop m.f1
    f1 := drop m.f2
    f2 := drop m.f3 }

abbrev V7 := Fin 7

def neg7 (x : V7) : V7 := ⟨6 - x.val, by omega⟩
def and7 (x y : V7) : V7 := ⟨min x.val y.val, by omega⟩
def or7  (x y : V7) : V7 := ⟨max x.val y.val, by omega⟩
def fus7 (x y : V7) : V7 := ⟨if x.val + y.val >= 6 then x.val + y.val - 6 else 0, by omega⟩
def imp7 (x y : V7) : V7 := ⟨min 6 (6 - x.val + y.val), by omega⟩

/-- Exact rational transition state (w_fwd_0, w_bwd_0, w_fwd_1, w_bwd_1) in [0, 1]^4
    represented by positive denominators and bounded numerators. -/
structure RationalChannelState where
  wf0_num : Nat
  wf0_den : Nat
  wb0_num : Nat
  wb0_den : Nat
  wf1_num : Nat
  wf1_den : Nat
  wb1_num : Nat
  wb1_den : Nat
  wf0_ok : wf0_den > 0 ∧ wf0_num ≤ wf0_den
  wb0_ok : wb0_den > 0 ∧ wb0_num ≤ wb0_den
  wf1_ok : wf1_den > 0 ∧ wf1_num ≤ wf1_den
  wb1_ok : wb1_den > 0 ∧ wb1_num ≤ wb1_den

/-- Genuine physical time-reversal involution T : (W_fwd, W_bwd) |-> (W_bwd, W_fwd)
    swapping forward and backward conditional transition probabilities. -/
def timeReverseChannel (s : RationalChannelState) : RationalChannelState :=
  { wf0_num := s.wb0_num, wf0_den := s.wb0_den,
    wb0_num := s.wf0_num, wb0_den := s.wf0_den,
    wf1_num := s.wb1_num, wf1_den := s.wb1_den,
    wb1_num := s.wf1_num, wb1_den := s.wf1_den,
    wf0_ok := s.wb0_ok,
    wb0_ok := s.wf0_ok,
    wf1_ok := s.wb1_ok,
    wb1_ok := s.wf1_ok }

/-- Theorem: T is an exact involution on RationalChannelState (T(T(s)) = s). -/
theorem timeReverseChannel_involutive (s : RationalChannelState) :
    timeReverseChannel (timeReverseChannel s) = s := by
  cases s
  rfl

/-- Exact algebraic sign/magnitude quantization Q_7 directly from rational transition rates:
    - Boundary extremes: s_0 = wf0 * wb0 = 0 => 0 (F_abs); s_0 = 1 => 6 (T_abs).
    - Interior modes compare the cross-ratio R = (chi_0 / chi_1) = (wf0 * wf1) / (wb0 * wb1):
      LHS = wf0_num * wf1_num * wb0_den * wb1_den
      RHS = wb0_num * wb1_num * wf0_den * wf1_den
      - LHS = RHS => 3 (0_O, exact sectorial equilibrium chi_0 = chi_1)
      - LHS > RHS => 4 (U_fwd, moderate forward expansion) or 5 (T_rel, strong >= 2x drift)
      - LHS < RHS => 2 (U_bwd, moderate backward contraction) or 1 (F_rel, strong <= 1/2x drift) -/
def quantizeRationalQ7 (s : RationalChannelState) : V7 :=
  let s0_num := s.wf0_num * s.wb0_num
  let s0_den := s.wf0_den * s.wb0_den
  if s0_num == 0 then ⟨0, by decide⟩
  else if s0_num == s0_den then ⟨6, by decide⟩
  else
    let lhs := s.wf0_num * s.wf1_num * s.wb0_den * s.wb1_den
    let rhs := s.wb0_num * s.wb1_num * s.wf0_den * s.wf1_den
    if lhs == rhs then ⟨3, by decide⟩
    else if lhs > rhs then
      if lhs ≥ 2 * rhs then ⟨5, by decide⟩ else ⟨4, by decide⟩
    else
      if rhs ≥ 2 * lhs then ⟨1, by decide⟩ else ⟨2, by decide⟩

/-- Universal Theorem: For ALL valid interior rational channel states (0 < s_0 < 1),
    Q_7 is strictly Z_2-equivariant under physical time-reversal:
    Q_7(T(s)) = ~Q_7(s) = 6 - Q_7(s). -/
