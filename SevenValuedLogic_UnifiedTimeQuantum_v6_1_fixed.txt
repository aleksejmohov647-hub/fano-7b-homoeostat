/-- Structural classification of any transformation f in <a, b> into its exact
    Markov lumpability class (0..11) by image multiset / cyclic gap geometry. -/
def lumpClass (f : Trans4) : Nat :=
  let r := rankTrans4 f
  if r == 1 then 11
  else if r == 4 then 0
  else if r == 3 then
    let counts := allStates.map fun s =>
      ([f.1, f.2.1, f.2.2.1, f.2.2.2].filter (· == s)).length
    let m := (allStates.find? fun s => counts.getD s.val 0 == 0).getD 0
    let d := (allStates.find? fun s => counts.getD s.val 0 == 2).getD 0
    let gap := (d.val + 4 - m.val) % 4
    let pre := allStates.filter fun s => evalTrans4 f s == d
    let p0 := (pre.getD 0 0).val
    let p1 := (pre.getD 1 0).val
    let span := (p1 + 4 - p0) % 4
    if span == 1 || span == 3 then
      if gap == 1 then 1 else if gap == 2 then 2 else 3
    else
      if gap == 1 then 2 else if gap == 2 then 4 else 1
