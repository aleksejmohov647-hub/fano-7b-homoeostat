structure Perm (n : Nat) where
  toFun : Fin n → Fin n
  invFun : Fin n → Fin n
  left_inv : ∀ x, invFun (toFun x) = x
  right_inv : ∀ x, toFun (invFun x) = x

attribute [inline] Perm.toFun Perm.invFun

namespace Perm

@[ext]
theorem ext {n : Nat} {p q : Perm n} (h : ∀ x, p.toFun x = q.toFun x) : p = q := by
  cases p with
  | mk f fi fl fr =>
    cases q with
    | mk g gi gl gr =>
      have h1 : f = g := funext h
      subst h1
      have h2 : fi = gi := by
        funext x
        calc
          fi x = fi (f (gi x)) := by rw [gr]
          _ = gi x := fl (gi x)
      subst h2
      rfl

@[inline] def one (n : Nat) : Perm n where
  toFun := id
  invFun := id
  left_inv _ := rfl
  right_inv _ := rfl

@[inline] def mul {n : Nat} (p q : Perm n) : Perm n where
  toFun x := p.toFun (q.toFun x)
  invFun x := q.invFun (p.invFun x)
  left_inv x := by simp [p.left_inv, q.left_inv]
  right_inv x := by simp [q.right_inv, p.right_inv]

@[inline] def inv {n : Nat} (p : Perm n) : Perm n where
  toFun := p.invFun
  invFun := p.toFun
  left_inv := p.right_inv
  right_inv := p.left_inv

instance (n : Nat) : Inhabited (Perm n) := ⟨one n⟩
instance (n : Nat) : One (Perm n) := ⟨one n⟩
instance (n : Nat) : Mul (Perm n) := ⟨mul⟩
instance (n : Nat) : Inv (Perm n) := ⟨inv⟩

@[inline] def swap {n : Nat} (i j : Fin n) : Perm n where
  toFun x := if x == i then j else if x == j then i else x
  invFun x := if x == i then j else if x == j then i else x
  left_inv x := by
    by_cases hxi : x = i
    · subst hxi
      simp
    · by_cases hxj : x = j
      · subst hxj
        have hji : (j == i) = false := beq_false_of_ne (Ne.symm hxi)
        simp [hji]
      · have h1 : (x == i) = false := beq_false_of_ne hxi
        have h2 : (x == j) = false := beq_false_of_ne hxj
        simp [h1, h2]
  right_inv x := by
    by_cases hxi : x = i
    · subst hxi
      simp
    · by_cases hxj : x = j
      · subst hxj
        have hji : (j == i) = false := beq_false_of_ne (Ne.symm hxi)
        simp [hji]
      · have h1 : (x == i) = false := beq_false_of_ne hxi
        have h2 : (x == j) = false := beq_false_of_ne hxj
        simp [h1, h2]

/-- Exact algebraic sign/magnitude quantization Q_7 directly from rational transition rates. -/
@[inline] def quantizeRationalQ7 (s : RationalChannelState) : V7 :=
  let s0_num := s.wf0_num * s.wb0_num
  let s0_den := s.wf0_den * s.wb0_den
  if s0_num == 0 then ⟨0, by decide⟩
  else if s0_num == s0_den then ⟨6, by decide⟩
  else
    let lhs := s.wf0_num * s.wf1_num * s.wb0_den * s.wb1_den
    let rhs := s.wb0_num * s.wb1_num * s.wf0_den * s.wf1_den
    if lhs == rhs then ⟨3, by decide⟩
    else if lhs > rhs then
      if lhs ≥ 2 * rhs then ⟨5, by decide⟩ else ⟨4, by decide⟩
    else
      if rhs ≥ 2 * lhs then ⟨1, by decide⟩ else ⟨2, by decide⟩

/-- Universal Theorem: For ALL valid interior rational channel states (0 < s_0 < 1),
    Q_7 is strictly Z_2-equivariant under physical time-reversal:
    Q_7(T(s)) = ~~~Q_7(s) = 6 - Q_7(s). -/
theorem quantizeRationalQ7_equivariant (s : RationalChannelState)
    (h_interior : s.wf0_num * s.wb0_num > 0 ∧
                  s.wf0_num * s.wb0_num < s.wf0_den * s.wb0_den) :
    quantizeRationalQ7 (timeReverseChannel s) = ~~~(quantizeRationalQ7 s) := by
  have h1 : (s.wb0_num * s.wf0_num == 0) = false := by
    apply beq_false_of_ne
    have : s.wb0_num * s.wf0_num = s.wf0_num * s.wb0_num := Nat.mul_comm _ _
    omega
  have h2 : (s.wb0_num * s.wf0_num == s.wb0_den * s.wf0_den) = false := by
    apply beq_false_of_ne
    have hnum : s.wb0_num * s.wf0_num = s.wf0_num * s.wb0_num := Nat.mul_comm _ _
    have hden : s.wb0_den * s.wf0_den = s.wf0_den * s.wf0_den := Nat.mul_comm _ _
    omega
  have h3 : (s.wf0_num * s.wb0_num == 0) = false := by
    apply beq_false_of_ne; omega
  have h4 : (s.wf0_num * s.wb0_num == s.wf0_den * s.wf0_den) = false := by
    apply beq_false_of_ne; omega
  simp only [quantizeRationalQ7, timeReverseChannel, h1, h2, h3, h4, Bool.false_eq_true, ↓reduceIte,
    Complement.complement]
  set A := s.wf0_num * s.wf1_num * s.wb0_den * s.wb1_den
  set B := s.wb0_num * s.wb1_num * s.wf0_den * s.wf1_den
  by_cases hab : A = B
  · have hba : B = A := hab.symm
    simp [hab, hba, neg7]
  · have h_beq_ab : (A == B) = false := beq_false_of_ne hab
    have h_beq_ba : (B == A) = false := beq_false_of_ne (Ne.symm hab)
    simp only [h_beq_ab, h_beq_ba, Bool.false_eq_true, ↓reduceIte]
    by_cases hgt : A > B
    · have h_not_bgt : ¬ (B > A) := by omega
      simp only [hgt, h_not_bgt, ↓reduceIte]
      by_cases h2x : A ≥ 2 * B
      · simp [h2x, neg7]
      · simp [h2x, neg7]
    · have hbgt : B > A := by omega
      simp only [hgt, hbgt, ↓reduceIte]
      by_cases h2x : B ≥ 2 * A
      · simp [h2x, neg7]
      · simp [h2x, neg7]
